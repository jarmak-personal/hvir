+pub fn deinit(self: *Self) void {

+pub fn fail(self: *Self, code: Failure) void {
+    if (self.failure == .none) self.failure = code;

+fn entry(self: *Self, key: Key) ?std.AutoHashMapUnmanaged(Key, bool).GetOrPutResult {
+    if (self.failure != .none or key.generation >= self.cutoff) return null;
+    if (self.rows.getPtr(key)) |value| return .{ .value_ptr = value, .key_ptr = undefined, .found_existing = true };
+    if (self.rows.count() >= self.max_keys) {
+        self.fail(.bound);
+        return null;
+    }
+    return self.rows.getOrPut(self.alloc, key) catch {
+        self.fail(.out_of_memory);
+        return null;
+    };
+}
+pub fn write(self: *Self, generation: u64, row: u16) void {
+    const e = self.entry(.{ .generation = generation, .row = row }) orelse return;
+    if (e.found_existing and e.value_ptr.*) self.fail(.stale) else e.value_ptr.* = false;

+        return false;
+    }

+/// Resumable plaintext selection formatter. It shares the canonical codepoint
+/// encoder and preserves PageFormatter's pending blank/wrap accounting. The
+/// caller owns tracked pins and protects the PageList for each bounded step.
+pub const PlainCursor = struct {
+    start: *PageList.Pin,
+    end: *PageList.Pin,
+    next: *PageList.Pin,
+    rectangle: bool,
+    point: ?PageList.Pin = null,
+    point_offset: ?u64 = null,
+    total_bytes: u64 = 0,
+    phase: enum { prepare, scan, newline, emit, advance, done } = .prepare,
+    left: usize = 0,
+    right: usize = 0,
+    scan_x: usize = 0,
+    x: usize = 0,
+    cp_index: usize = 0,
+    cell_started: bool = false,
+    blank_rows: usize = 0,
+    blank_cells: usize = 0,
+    newlines: usize = 0,
+    spaces: usize = 0,
+    pub const Step = struct { written: usize, visited: usize, work: usize, done: bool };
+    pub fn step(self: *PlainCursor, out: []u8, budget: usize) Step {
+        var written: usize = 0;
+        var visited: usize = 0;
+        var work: usize = 0;
+        while (written < out.len and work < budget) {
+            const page_ = self.next.node.page();
+            const row = page_.getRow(self.next.y);
+            const cells = page_.getCells(row);
+            switch (self.phase) {
+                .done => break,
+                .prepare => {
+                    work += 1;
+                    const first = self.next.node == self.start.node and self.next.y == self.start.y;
+                    const last = self.next.node == self.end.node and self.next.y == self.end.y;
+                    self.left = if (self.rectangle or first) @min(self.start.x, cells.len - 1) else 0;
+                    self.right = if (self.rectangle or last) @min(@as(usize, self.end.x) + 1, cells.len) else cells.len;
+                    if (self.left > 0) switch (cells[self.left].wide) {
+                        .spacer_tail => self.left -= 1,
+                        .spacer_head => {
+                            self.phase = .advance;
+                            continue;
+                        },
+                        else => {},
+                    };
+                    self.scan_x = self.left;
+                    self.phase = .scan;
+                },
+                .scan => {
+                    if (self.scan_x >= self.right) {
+                        self.blank_rows += 1;
+                        self.phase = .advance;
+                        continue;
+                    }
+                    const cell = cells[self.scan_x];
+                    self.scan_x += 1;
+                    visited += 1;
+                    work += 1;
+                    if (!cell.hasText()) continue;
+                    self.newlines = self.blank_rows;
+                    self.blank_rows = if (!row.wrap) 1 else 0;
+                    if (!row.wrap_continuation) self.blank_cells = 0;
+                    self.x = self.left;
+                    self.phase = .newline;
+                },
+                .newline => {
+                    const count = @min(self.newlines, @min(out.len - written, budget - work));
+                    @memset(out[written..][0..count], '\n');
+                    written += count;
+                    self.total_bytes += count;
+                    self.newlines -= count;
+                    work += count;
+                    if (self.newlines == 0) self.phase = .emit;
+                },
+                .emit => {
+                    if (self.spaces > 0) {
+                        const count = @min(self.spaces, @min(out.len - written, budget - work));
+                        @memset(out[written..][0..count], ' ');
+                        written += count;
+                        self.total_bytes += count;
+                        self.spaces -= count;
+                        work += count;
+                        continue;
+                    }
+                    if (self.x >= self.right) {
+                        self.phase = .advance;
+                        continue;
+                    }
+                    const cell = &cells[self.x];
+                    if (!self.cell_started) {
+                        visited += 1;
+                        work += 1;
+                        if (cell.wide == .spacer_head or cell.wide == .spacer_tail) {
+                            self.x += 1;
+                            continue;
+                        }
+                        if (!cell.hasText()) {
+                            self.blank_cells += 1;
+                            self.x += 1;
+                            continue;
+                        }
+                        self.cell_started = true;
+                        self.cp_index = 0;
+                        self.spaces = self.blank_cells;
+                        self.blank_cells = 0;
+                        if (self.point) |point| if (point.node == self.next.node and point.y == self.next.y and point.x == self.x) {
+                            self.point_offset = self.total_bytes + self.spaces;
+                        };
+                        if (self.spaces > 0) continue;
+                    }
+                    const grapheme: []const u21 = if (cell.content_tag == .codepoint_grapheme)
+                        page_.lookupGrapheme(cell).?
+                    else
+                        &.{};
+                    if (self.cp_index > grapheme.len) {
+                        self.cell_started = false;
+                        self.x += 1;
+                        continue;
+                    }
+                    const cp = if (self.cp_index == 0) cell.codepoint() else grapheme[self.cp_index - 1];
+                    var encoded: [512]u8 = undefined;
+                    var count: usize = 0;
+                    _ = PageFormatter.encodeCodepoint(.plain, &encoded, &count, cp);
+                    if (count > out.len - written) break;
+                    @memcpy(out[written..][0..count], encoded[0..count]);
+                    written += count;
+                    self.total_bytes += count;
+                    self.cp_index += 1;
+                    work += 1;
+                },
+                .advance => {
+                    work += 1;
+                    if (self.next.node == self.end.node and self.next.y == self.end.y) {
+                        self.phase = .done;
+                        continue;
+                    }
+                    self.next.* = self.next.down(1) orelse {
+                        self.phase = .done;
+                        continue;
+                    };
+                    self.phase = .prepare;
+                    break; // Caller authenticates/captures the next row before reading it.
+                },
+            }
+        }
+        return .{ .written = written, .visited = visited, .work = work, .done = self.phase == .done };
+    }
