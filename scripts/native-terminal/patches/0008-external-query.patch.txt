+fn allocateRowId() !u64 {
+    var current = next_row_id.load(.monotonic);
+    while (current != std.math.maxInt(u64)) {
+        if (next_row_id.cmpxchgWeak(current, current + 1, .monotonic, .monotonic)) |actual| current = actual else return current;
+    }

+/// Allocation is separate from initialization: the query driver charges and
+/// yields while prepare initializes the actual native row capacity.

+pub fn create(self: *Self, columns: usize, location: Location) !*Row {

+    const id = try allocateRowId();
+    const row = try self.alloc.create(Row);
+    errdefer self.alloc.destroy(row);
+    const cells = try self.alloc.alloc(Cell, try std.math.add(usize, columns, 1));
+    errdefer self.alloc.free(cells);
+    const maxima = try self.alloc.alloc(u64, cells.len + 1);
+    errdefer self.alloc.free(maxima);

+pub fn prepare(_: *Self, row: *Row, budget: usize) usize {
+    const count = @min(budget, row.cells.len - row.prepared);
+    const stop = row.prepared + count;
+    for (row.cells[row.prepared..stop]) |*cell| cell.* = .{ .written = row.pending_stamp };

+/// Certify a contiguous cell interval in <=2*usizeBits metadata operations.
+/// Counts returned by this method are charged by the bounded query driver.
+pub fn unchanged(self: *Self, row: *const Row, start: usize, end: usize, cutoff: u64, work: *usize) bool {

+        work.* += 1;

+            cursor -= 1;
+        }

+    while (result.written < limit and budget - result.work >= cost) {
+        const target = (self.head + self.length) % self.bytes.len;
+        self.bytes[target] = bytes[result.written];
+        self.points[target] = points[result.written];
+        result.work += 1;
+        var node = self.max_base + target;
+        self.maxima[node] = points[result.written].written;
+        result.work += 1;
+        while (node > 1) {
+            node /= 2;
+            self.maxima[node] = @max(self.maxima[node * 2], self.maxima[node * 2 + 1]);
+            result.work += 1;
+        }
+        result.written += 1;
+        self.length += 1;
+    }
+    return result;
+}
+fn maximum(self: *const Self, start: usize, end: usize, work: *usize) u64 {
+    var left = start + self.max_base;
+    var right = end + self.max_base;
+    var result: u64 = 0;
+    while (left < right) {
+        work.* += 1;
+        if (left & 1 != 0) {
+            result = @max(result, self.maxima[left]);
+            left += 1;
+            work.* += 1;
+        }
+        if (right & 1 != 0) {
+            right -= 1;
+            result = @max(result, self.maxima[right]);
+            work.* += 1;
+        }
+        left /= 2;
+        right /= 2;
+    }
+    return result;
+}
+fn point(self: *const Self, offset: usize) Point {
+    std.debug.assert(offset < self.length);
+    return self.points[(self.head + offset) % self.points.len];

+pub fn step(self: *Self, needle: []const u8, sensitive: bool, budget: usize) Step {
+    const reserve = 8 * (self.depth() + 1) + 2;
+    var work: usize = 0;
+    while (budget - work > reserve) {
+        const first = @min(self.length, self.bytes.len - self.head);
+        const result = self.matcher.step(.{ self.bytes[self.head..][0..first], self.bytes[0 .. self.length - first] }, needle, sensitive, budget - work - reserve);
+        work += result.work;
+        var match: ?Match = null;
+        if (result.match) |offset| {
+            const start_point = self.point(offset);
+            const end_point = self.point(offset + needle.len - 1);
+            work += 2;
+            if (start_point.first and end_point.last) {
+                const start = (self.head + offset) % self.bytes.len;
+                const count = @min(needle.len, self.bytes.len - start);
+                const max_written = @max(self.maximum(start, start + count, &work), self.maximum(0, needle.len - count, &work));
+                match = .{ .start = start_point, .end = end_point, .max_written = max_written };
+            }
+        }
+        const removed = self.matcher.offset - self.matcher.matched;
+        self.head = (self.head + removed) % self.bytes.len;
+        self.length -= removed;
+        self.matcher.rebase(removed);
+        if (match != null or result.needs_data) return .{ .work = work, .match = match, .needs_data = result.needs_data };
+        if (result.work == 0) break;
+    }
+    return .{ .work = work, .match = null, .needs_data = false };

+//! window retains bytes/point mapping; this owns only the needle's failure table.
+const std = @import("std");
+const Self = @This();
+failure: []usize,
+prepared: usize = 0,
+prepare_matched: usize = 0,
+prepare_write: ?usize = null,
+offset: usize = 0,
+matched: usize = 0,
+
+pub const Result = struct { work: usize = 0, match: ?usize = null, needs_data: bool = false };
+
+pub fn init(alloc: std.mem.Allocator, length: usize) !Self {
+    return .{ .failure = try alloc.alloc(usize, length) };

+pub fn deinit(self: *Self, alloc: std.mem.Allocator) void {
+    alloc.free(self.failure);
+    self.* = undefined;

+fn equal(a: u8, b: u8, sensitive: bool) bool {
+    return if (sensitive) a == b else std.ascii.toLower(a) == std.ascii.toLower(b);
+}
+/// Query preparation runs outside the terminal lock. A comparison or failure
+/// table initialization is one work unit; even the long-prefix fallback yields.
+pub fn prepare(self: *Self, needle: []const u8, sensitive: bool, budget: usize) usize {
+    std.debug.assert(needle.len == self.failure.len);
+    var work: usize = 0;
+    while (self.prepared < needle.len and work < budget) {
+        if (self.prepared == 0) {
+            self.failure[0] = 0;
+            self.prepared = 1;
+            work += 1;
+            continue;
+        }
+        if (self.prepare_write) |value| {
+            self.failure[self.prepared] = value;
+            self.prepared += 1;
+            self.prepare_matched = value;
+            self.prepare_write = null;
+            work += 1;
+            continue;
+        }
+        work += 1;
+        if (equal(needle[self.prepared], needle[self.prepare_matched], sensitive)) {
+            self.prepare_write = self.prepare_matched + 1;
+        } else if (self.prepare_matched > 0) {
+            self.prepare_matched = self.failure[self.prepare_matched - 1];
+        } else {
+            self.prepare_write = 0;
+        }
+    }
+    return work;
+}
+/// Search both slices of a circular window, without copying the overlap. Each
+/// byte comparison (including failure-link fallback) is actual charged work.
+pub fn step(self: *Self, slices: [2][]const u8, needle: []const u8, sensitive: bool, budget: usize) Result {
+    std.debug.assert(self.prepared == needle.len);
+    var result: Result = .{};
+    const len = slices[0].len + slices[1].len;
+    if (needle.len == 0) {
+        result.needs_data = true;
+        return result;
+    }
+    while (self.offset < len and result.work < budget) {
+        const actual = if (self.offset < slices[0].len) slices[0][self.offset] else slices[1][self.offset - slices[0].len];
+        result.work += 1;
+        if (equal(actual, needle[self.matched], sensitive)) {
+            self.offset += 1;
+            self.matched += 1;
+            if (self.matched == needle.len) {
+                result.match = self.offset - needle.len;
+                self.matched = self.failure[self.matched - 1];
+                return result;
+            }
+        } else if (self.matched > 0) {
+            self.matched = self.failure[self.matched - 1];
+        } else {
+            self.offset += 1;
+        }
+    }
+    result.needs_data = self.offset == len;
+    return result;
+}

+//! Query-owned opaque occurrence IDs. This is metadata, not a text index.
+//! Stable segmented node slabs avoid rehash/copies; retired tree nodes are
+//! reused, so writes do not append a lifetime journal. No terminal pins live here.
+const Self = @This();
+const SegmentedList = @import("../../datastruct/segmented_list.zig").SegmentedList;
+const std = @import("std");
+const Record = @import("QueryResults.zig").Match;
+const Node = union(enum) {
+    leaf: Leaf,
+    branch: struct { bit: u6, children: [2]?*Node },
+    unused: ?*Node,
+};
+pub const Leaf = struct { value: Record, index: usize, seen: u64, previous_index: usize = 0, previous_seen: u64 = 0 };
+alloc: std.mem.Allocator,
+root: ?*Node = null,
+nodes: SegmentedList(Node, 0) = .{},
+free: ?*Node = null,
+count: usize = 0,
+fn direction(id: u64, bit: u6) usize {
+    return @intCast((id >> bit) & 1);
+}
+fn take(self: *Self, work: *usize) !*Node {
+    if (self.free) |node| {
+        self.free = node.unused;
+        work.* += 1;
+        return node;
+    }
+    if (self.nodes.len == std.math.maxInt(usize)) return error.OutOfMemory;
+    const shelves = @bitSizeOf(usize) - @clz(self.nodes.len + 1);
+    const cost = 1 + (if (shelves > self.nodes.dynamic_segments.len) self.nodes.dynamic_segments.len + 4 else @as(usize, 0));
+    const node = try self.nodes.addOne(self.alloc);
+    work.* += cost;
+    return node;
+}
+fn put(self: *Self, node: *Node) void {
+    node.* = .{ .unused = self.free };
+    self.free = node;
+}
+pub fn deinit(self: *Self) void {
+    self.nodes.deinit(self.alloc);
+    self.* = undefined;
+}
+pub fn get(self: *Self, id: u64, work: *usize) ?*Leaf {
+    var node = self.root orelse return null;
+    while (node.* == .branch) {
+        work.* += 1;
+        node = node.branch.children[direction(id, node.branch.bit)].?;
+    }
+    work.* += 1;
+    return if (node.leaf.value.occurrence == id) &node.leaf else null;
+}
+/// Caller reserves208 units before this finite <=64-bit pair of walks.
+/// The result record is copied into this owner, never a borrowed page/match.
+pub fn insert(self: *Self, value: Record, index: usize, pass: u64, work: *usize) !void {
+    var found = self.root;
+    while (found) |node| {
+        work.* += 1;
+        if (node.* == .leaf) break;
+        found = node.branch.children[direction(value.occurrence, node.branch.bit)];
+    }
+    if (found) |node| if (node.leaf.value.occurrence == value.occurrence) {
+        if (node.leaf.seen != pass) {
+            node.leaf.previous_index = node.leaf.index;
+            node.leaf.previous_seen = node.leaf.seen;
+        }
+        node.leaf.value = value;
+        node.leaf.index = index;
+        node.leaf.seen = pass;
+        work.* += 1;
+        return;
+    };
+    const leaf = try self.take(work);
+    errdefer self.put(leaf);
+    if (found) |node| {
+        const bit: u6 = @intCast(63 - @clz(value.occurrence ^ node.leaf.value.occurrence));
+        const branch = try self.take(work);
+        var slot = &self.root;
+        while (slot.*.?.* == .branch and slot.*.?.branch.bit > bit) {
+            work.* += 1;
+            slot = &slot.*.?.branch.children[direction(value.occurrence, slot.*.?.branch.bit)];
+        }
+        const side = direction(value.occurrence, bit);
+        var children: [2]?*Node = undefined;
+        children[side] = leaf;
+        children[side ^ 1] = slot.*;
+        branch.* = .{ .branch = .{ .bit = bit, .children = children } };
+        slot.* = branch;
+        work.* += 1;
+    } else self.root = leaf;
+    leaf.* = .{ .leaf = .{ .value = value, .index = index, .seen = pass } };
+    self.count += 1;
+    work.* += 1;
+}
+pub fn retire(self: *Self, id: u64, work: *usize) bool {
+    var slot = &self.root;
+    var parent: ?*?*Node = null;
+    while (slot.*) |node| {
+        work.* += 1;
+        if (node.* == .leaf) break;
+        parent = slot;
+        slot = &node.branch.children[direction(id, node.branch.bit)];
+    }
+    const leaf = slot.* orelse return false;
+    if (leaf.leaf.value.occurrence != id) return false;
+    if (parent) |p| {
+        const branch = p.*.?;
+        const side = direction(id, branch.branch.bit);
+        p.* = branch.branch.children[side ^ 1];
+        self.put(branch);
+    } else self.root = null;
+    self.put(leaf);
+    self.count -= 1;
+    work.* += 2;
+    return true;
+}

+pub fn createWithOptions(alloc: std.mem.Allocator, t: *Terminal, needle: []const u8, sensitive: bool, options: Options) !*Self {
+    const screen = t.screens.get(.primary).?;
+    const pages = &screen.pages;
+    const self = try alloc.create(Self);
+    errdefer alloc.destroy(self);

+fn loadPending(self: *Self) !void {

+    if (next.* == std.math.maxInt(u64)) return error.IdentityUnavailable;
+    const id = next.*;

+fn reconcile(self: *Self, t: *Terminal, match: Window.Match, budget: usize, work: *usize) !?Results.Match {
+    const prior = self.options.prior orelse return try self.fresh(match);
+    const identity = t.screens.get(.primary).?.pages.text_identity.?;
+    while (self.prior_index < prior.results.len()) {
+        if (budget - work.* < 66) return null;
+        const old = prior.results.at(self.prior_index).?.*;
+        const row = identity.getWithWork(old.start.row, work) orelse {
+            self.prior_index += 1;
+            work.* += 1;
+            continue;
+        };
+        // An unvisited retained old start lies later in this forward pass.
+        if (row.observed_pass != self.options.pass) return try self.fresh(match);
+        if (row.observed_row < match.start.observed_row or
+            (row.observed_row == match.start.observed_row and old.start.column < match.start.column))
+        {
+            self.prior_index += 1;
+            work.* += 1;
+            continue;
+        }
+        if (old.start.row == match.start.row and old.start.column == match.start.column) {
+            self.prior_index += 1;
+            work.* += 1;
+            if (old.end.row == match.end.row and old.end.column == match.end.column and match.max_written <= old.cutoff)
+                return .{ .occurrence = old.occurrence, .start = match.start, .end = match.end, .cutoff = old.cutoff };
+        }
+        return try self.fresh(match);
+    }
+    return try self.fresh(match);

+pub fn step(self: *Self, t: *Terminal, budget: usize) !Step {
+    var result: Step = .{};
+    defer self.last_step = result;
+    if (self.done) return .{ .done = true };
+    if (t.screens.generation(.primary) != self.generation) return error.Stale;
+    // Eviction can remove a captured prefix or even the unread finite extent.
+    // It does not close the query or invalidate unaffected retained occurrence
+    // IDs. Complete this pass; the changed source snapshot schedules its next
+    // finite reconciliation. Resolution still authenticates every cell.
+    if (self.end.garbage or self.next.garbage) {
+        self.done = true;
+        return .{ .work = 1, .done = true };
+    }
+    const pages = &t.screens.get(.primary).?.pages;
+    while (result.work < budget) {
+        if (!self.window.prepared()) {
+            result.work += self.window.prepare(self.needle, self.sensitive, budget - result.work);
+            continue;
+        }
+        if (self.pending_match) |match| {
+            // A finite pass can observe writes newer than its authority fence.
+            // Defer those candidates to the next source reconciliation, rather
+            // than publish an occurrence that its own cutoff cannot validate.
+            result.work += 1;
+            if (match.max_written > self.cutoff) {
+                self.pending_match = null;
+                continue;
+            }
+            if (self.pending_record == null) {
+                const record = try self.reconcile(t, match, budget, &result.work) orelse break;
+                self.pending_record = record;
+            }
+            // Tree insertion <=208 metadata units; native shelf growth <=69.
+            if (budget - result.work < 280) break;
+            const record = self.pending_record.?;
+            if (self.options.registry) |registry|
+                try registry.insert(record, self.results.len(), self.options.pass, &result.work);
+            if (!try self.results.append(record, budget, &result.work)) break;
+            self.revision += 1;
+            self.pending_match = null;
+            self.pending_record = null;
+            continue;
+        }
+        const found = self.window.step(self.needle, self.sensitive, budget - result.work);
+        result.work += found.work;
+        if (found.match) |match| {
+            self.pending_match = match;
+            continue;
+        }
+        if (!found.needs_data) break;
+        if (self.source_done) {
+            self.done = true;
+            result.done = true;
+            break;
+        }
+        if (!self.next.node.restoreBounded(budget, &result.work)) break;
+        var pending_ready = true;
+        for (0..2) |i| if (self.pending_present[i]) {
+            if (self.pending_pins[i].garbage) {
+                self.done = true;
+                result.done = true;
+                result.work += 1;
+                return result;
+            }
+            if (!self.pending_pins[i].node.restoreBounded(budget, &result.work)) {
+                pending_ready = false;
+                break;
+            }
+        };
+        if (!pending_ready) break;
+        const row = try pages.prepareTextIdentityRow(self.next.*, budget, &result.work) orelse break;
+        row.observed_row = self.source_row;
+        row.observed_pass = self.options.pass;
+        // UTF-8 output can contain four bytes per codepoint. Reserve conversion
+        // and window-copy work before invoking the formatter, rather than
+        // producing an uncharged suffix or parking borrowed cell pointers.
+        const allowance = @min(@as(usize, 128), (budget - result.work) / 128);
+        if (allowance == 0) break;
+        var bytes: [512]u8 = undefined;
+        var pins: [512]PageList.Pin = undefined;
+        var boundaries: [512]u2 = undefined;
+        var points: [512]Window.Point = undefined;
+        self.loadPending() catch {
+            self.done = true;
+            result.done = true;
+            result.work += 1;
+            return result;
+        };
+        const before = self.next.*;
+        const formatted = self.formatter.stepMappedBoundaries(&bytes, &pins, &boundaries, allowance);
+        if (self.next.node != before.node or self.next.y != before.y) self.source_row += 1;
+        self.savePending();
+        result.work += formatted.work;
+        result.visited += formatted.visited;
+        for (pins[0..formatted.written], 0..) |pin, i| {
+            const identity_row = pages.textIdentityRow(pin) orelse return error.Stale;
+            points[i] = .{ .row = identity_row.id, .column = pin.x, .observed_row = identity_row.observed_row, .written = identity_row.cells[pin.x].written, .boundary = pin.x == pin.node.cols(), .first = boundaries[i] & 1 != 0, .last = boundaries[i] & 2 != 0 };
+            result.work += 1;
+        }
+        const copied = self.window.append(bytes[0..formatted.written], points[0..formatted.written], budget - result.work);
+        std.debug.assert(copied.written == formatted.written);
+        result.work += copied.work;
+        self.source_done = formatted.done;
+        if (formatted.work == 0 and copied.written == 0) {
+            // A prior restore/preparation may leave less than one mapped
+            // codepoint's work. Yield the work already performed; the next
+            // full-budget call resumes this exact formatter state.
+            if (result.work != 0) break;
+            return error.NoProgress;
+        }
+    }
+    return result;
+}
